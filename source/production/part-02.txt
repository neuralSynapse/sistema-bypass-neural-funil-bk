  function calculateResult(){
    const totals = Object.fromEntries(AXES.map(a => [a,0]));
    const weighted = Object.fromEntries(AXES.map(a => [a,0]));
    const manifestations = Object.fromEntries(AXES.map(a => [a,[]]));
    steps.filter(s => s.type === 'question').forEach(question => {
      const option = question.options.find(x => x.value === state.answers[question.id]);
      if (!option) return;
      AXES.forEach(axis => {
        const value = Number(option.scores?.[axis] || 0);
        totals[axis] += value;
        weighted[axis] += value * Number(question.weight || 1);
        const text = option.manifestations?.[axis];
        if (value > 0 && text) manifestations[axis].push({ value, weight:question.weight||1, text });
      });
    });
    const ranking = [...AXES].sort((a,b) => {
      if (totals[b] !== totals[a]) return totals[b] - totals[a];
      if (weighted[b] !== weighted[a]) return weighted[b] - weighted[a];
      return AXIS_PRIORITY.indexOf(a) - AXIS_PRIORITY.indexOf(b);
    });
    const totalSeverity = AXES.reduce((sum,a)=>sum+totals[a],0);
    const noDominant = totalSeverity === 0;
    if (noDominant) {
      const goalAxis = { melhor_diagnostico:'diagnosticoFraco', melhor_valor:'valorDifuso', melhor_objecao:'objecaoReativa', melhor_fechamento:'fechamentoAberto' }[state.answers.goal] || 'diagnosticoFraco';
      const idx = ranking.indexOf(goalAxis);
      if (idx > 0) ranking.splice(idx,1);
      ranking.unshift(goalAxis);
    }
    let level;
    if (totalSeverity >= 22) level = ['Estrutura Inicial','A conversa depende bastante de improvisação e perde direção em vários momentos.'];
    else if (totalSeverity >= 12) level = ['Estrutura Parcial','Existem boas práticas, mas elas ainda não formam uma sequência consistente.'];
    else if (totalSeverity >= 5) level = ['Estrutura Funcional','Existe condução razoavelmente clara, com rupturas localizadas.'];
    else level = ['Estrutura Consolidada','A conversa possui boa sequência, com pontos pontuais de otimização.'];
    const sortedMan = manifestations[ranking[0]].sort((a,b)=>(b.value*b.weight)-(a.value*a.weight)).map(x=>x.text);
    return { totals, weighted, primary:ranking[0], secondary:ranking[1], totalSeverity, noDominant, level, manifestations:sortedMan.slice(0,3) };
  }

  function phaseFor(step){
    if (step <= 1) return 'Entrada';
    if (step <= 6) return 'Contexto';
    if (step <= 16) return 'Conversa';
    if (step <= 22) return 'Objeções';
    if (step <= 24) return 'Análise';
    if (step === 25) return 'Resultado';
    return 'Sistema';
  }

  function updateChrome(){
    const progress = Math.max(0, Math.min(100, (state.step / (steps.length-1))*100));
    progressBar.style.width = `${progress}%`;
    progressWrap.hidden = state.step === 0;
    phaseLabel.textContent = phaseFor(state.step);
    bottomNav.hidden = state.step === 0 || state.step >= 27;
    backButton.disabled = state.step <= 0;
  }

  function go(next, {push=true}={}){
    state.step = Math.max(0, Math.min(steps.length-1, next));
    saveState();
    if (push) history.pushState({bkStep:state.step},'',`#etapa-${state.step+1}`);
    render();
  }

  function answerQuestion(question, value, customText=''){
    state.answers[question.id] = value;
    if (question.id === 'sellType') {
      state.context.sellType = value;
      if (value === 'outro') state.context.sellTypeOther = customText.trim();
    }
    if (question.id === 'channel') state.context.channel = value;
    if (question.id === 'ticket') state.context.ticket = value;
    saveState();
    trackAction('BK_Answer',{question_id:question.id,answer:value});
    if (question.completeEvent) fireOnce(question.completeEvent,{question_id:question.id});
    setTimeout(() => go(state.step+1), 150);
  }

  function renderQuestion(question){
    const current = state.answers[question.id];
    screen.innerHTML = `<div class="bk-frame">
      <div class="bk-question-head"><div class="bk-kicker">${escapeHtml(question.kicker)}</div><h1 class="bk-question">${escapeHtml(question.question)}</h1><p class="bk-helper">${escapeHtml(question.helper)}</p></div>
      <div class="bk-options" role="group" aria-label="${escapeHtml(question.question)}">
        ${question.options.map((option,i)=>`<button type="button" class="bk-option" data-value="${escapeHtml(option.value)}" aria-pressed="${current===option.value}"><span class="bk-letter">${String.fromCharCode(65+i)}</span><span>${escapeHtml(option.label)}</span><span class="bk-arrow">→</span></button>`).join('')}
      </div>
      ${question.allowOther ? `<div class="bk-other" id="otherBox" ${current==='outro'?'':'hidden'}><label for="otherInput" class="bk-helper">Descreva em poucas palavras</label><input class="bk-other-input" id="otherInput" maxlength="80" placeholder="${escapeHtml(question.otherPlaceholder||'Outro')}"><button class="bk-btn bk-btn-primary" id="otherContinue" type="button">Continuar</button></div>`:''}
    </div>`;
    screen.querySelectorAll('.bk-option').forEach(btn => btn.addEventListener('click',()=>{
      const value = btn.dataset.value;
      screen.querySelectorAll('.bk-option').forEach(b => b.setAttribute('aria-pressed','false'));
      btn.setAttribute('aria-pressed','true');
      if (question.allowOther && value === 'outro') {
        state.answers[question.id] = value; saveState();
        const box = document.getElementById('otherBox'); box.hidden=false; document.getElementById('otherInput').focus();
      } else answerQuestion(question,value);
    }));
    if (question.allowOther) {
      const input = document.getElementById('otherInput');
      if (current==='outro' && state.context.sellTypeOther) input.value = state.context.sellTypeOther;
      document.getElementById('otherContinue').addEventListener('click',()=>{
        const value=input.value.trim();
        if (!value){ input.focus(); return; }
        answerQuestion(question,'outro',value);
      });
    }
  }

  function render(){
    updateChrome();
    screen.classList.remove('bk-screen'); void screen.offsetWidth; screen.classList.add('bk-screen');
    const step = steps[state.step];
    if (!step) return;
    if (step.type === 'question') renderQuestion(step);
    else renderNarrative(step.type);
    main.focus({preventScroll:true});
    window.scrollTo({top:0,behavior:'instant'});
  }
